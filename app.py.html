def generate_gradcam(image_path: str, model_type: str) -> str:
    """Generates a Grad-CAM heatmap for the given image and returns it as a base64 string."""
    try:
        original_image = Image.open(image_path).convert("RGB")
        image_tensor = transform(original_image).unsqueeze(0).to(device)

        model = models_dict[model_type]

        if model_type == 'cnn':
            target_layers = [model.features[12]]
        elif model_type == 'resnet':
            target_layers = [model.layer4[-1]]
        elif model_type == 'vgg':
            target_layers = [model.features[28]]
        
        cam = GradCAM(model=model, target_layers=target_layers)
        
        # Generate the CAM
        grayscale_cam = cam(input_tensor=image_tensor, targets=None)
        grayscale_cam = grayscale_cam[0]
        
        # Prepare image for visualization
        display_image = original_image.resize((224, 224))
        display_image = np.array(display_image).astype(np.float32) / 255.0
        
        visualization = show_cam_on_image(display_image, grayscale_cam, use_rgb=True)
        
        # Convert to base64
        img_pil = Image.fromarray(visualization)
        buffered = io.BytesIO()
        img_pil.save(buffered, format="JPEG")
        img_str = base64.b64encode(buffered.getvalue()).decode("utf-8")
        
        return f"data:image/jpeg;base64,{img_str}"
    except Exception as e:
        print(f"GradCAM generation error: {e}")
        return ""

@app.post("/api/predict")
async def predict_mri(file: UploadFile = File(...), model_type: str = Form("cnn")):
    # Save the uploaded file temporarily
    file_ext = file.filename.split(".")[-1]
    temp_filename = f"results/{uuid.uuid4()}.{file_ext}"
    
    with open(temp_filename, "wb") as buffer:
        content = await file.read()
        buffer.write(content)
        
    try:
        # Run classification pipeline
        result = classify_mri(temp_filename, model_type)
        
        # Generate Grad-CAM
        gradcam_b64 = generate_gradcam(temp_filename, model_type)
        result["gradcam"] = gradcam_b64
        
        # Also return the original image as base64 for the frontend
        with open(temp_filename, "rb") as img_file:
            orig_b64 = base64.b64encode(img_file.read()).decode("utf-8")
            result["original_image_b64"] = f"data:image/{file_ext};base64,{orig_b64}"
            
        # Clean up temporary file
        os.remove(temp_filename)
        
        return JSONResponse(content=result)
    except Exception as e:
        if os.path.exists(temp_filename):
            os.remove(temp_filename)
        return JSONResponse(status_code=500, content={"error": str(e)})

class DiscussMonitoringRequest(BaseModel):
    patient_id: str
    report: dict
    chat_history: list = []
    user_message: str


@app.post("/api/monitoring/discuss")
async def discuss_monitoring(request: DiscussMonitoringRequest):
    if not llm_model:
        async def placeholder_stream():
            yield "Gemini API Key not configured. Unable to discuss the monitoring report."
        return StreamingResponse(placeholder_stream(), media_type="text/plain")

    # Construct chat context
    patient_id = request.patient_id
    report = request.report
    
    prompt = (
        f"You are the Report Agent, an AI medical assistant specialized in neuro-oncology longitudinal tracking.\n"
        f"You are discussing a brain tumor progression monitoring report for Patient ID: {patient_id}.\n\n"
        f"Here is the quantitative comparison report between the baseline and follow-up MRI scans:\n"
        f"```json\n"
        f"{json.dumps(report, indent=2)}\n"
        f"```\n\n"
        f"Analyze this data carefully. Note volume changes (growth/shrinkage), centroid shift (displacement), boundary surface area changes, shape sphericity, and affected brain regions.\n"
        f"Answer the user's question accurately, clearly, and in a professional, clinical yet compassionate tone. Avoid general medical disclaimers, but emphasize that this is a data-driven report.\n\n"
    )
    
    # Add history
    for msg in request.chat_history:
        role = msg.get("role", "user")
        content = msg.get("content", "")
        prompt += f"{role.capitalize()}: {content}\n"
    
    prompt += f"User: {request.user_message}\n"
    prompt += "Report Agent (AI): "
    
    async def generate_stream():
        try:
            response = llm_model.generate_content(prompt, stream=True)
            for chunk in response:
                if chunk.text:
                    yield chunk.text
                    await asyncio.sleep(0.01)
        except Exception as e:
            yield f"\n\n**Error generating discussion:** {str(e)}"
            
    return StreamingResponse(generate_stream(), media_type="text/plain")
