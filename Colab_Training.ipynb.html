## 1. Import Libraries and Load Data 
#%% 
import torch
import torch.nn as nn
import torch.optim as optim
from torchvision import datasets, transforms, models
from torch.utils.data import DataLoader
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.metrics import classification_report, confusion_matrix, accuracy_score, f1_score, precision_score, recall_score
import numpy as np
import time
import json
import os

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
print(f'Using Device: {device}')

# Exact same transforms as your dataloader.py
transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor()
])

# NOTE: Make sure these paths match where you extracted your dataset on Colab
train_dataset = datasets.ImageFolder(root='data/Training', transform=transform)
test_dataset = datasets.ImageFolder(root='data/Testing', transform=transform)

train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)
test_loader = DataLoader(test_dataset, batch_size=32, shuffle=False)

class_names = train_dataset.classes
print('Classes:', class_names)
#%% md 
## 2. Define the Model Architectures 
#%% 
# 1. Custom CNN (To evaluate your existing one if you upload the weights, or just as reference)
class BrainTumorCNN(nn.Module):
    def __init__(self):
        super(BrainTumorCNN, self).__init__()
        self.features = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),
            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),
            nn.Conv2d(128, 256, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2, 2)
        )
        self.classifier = nn.Sequential(
            nn.Linear(256 * 14 * 14, 512),
            nn.ReLU(),
            nn.Dropout(0.5),
            nn.Linear(512, 4)
        )
    def forward(self, x):
        x = self.features(x)
        x = x.view(x.size(0), -1)
        x = self.classifier(x)
        return x

# 2. ResNet18
def get_resnet():
    model = models.resnet18(pretrained=True)
    model.fc = nn.Linear(model.fc.in_features, 4)
    return model

# 3. VGG16
def get_vgg():
    model = models.vgg16(pretrained=True)
    # Freeze early layers to speed up training
    for param in model.features.parameters():
        param.requires_grad = False
    model.classifier[6] = nn.Linear(model.classifier[6].in_features, 4)
    return model
#%% md 
## 3. Training Function 
#%% 
def train_model(model, name, epochs=20, lr=0.001):
    model = model.to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(filter(lambda p: p.requires_grad, model.parameters()), lr=lr)
    best_acc = 0.0
    print(f'\n--- Training {name} ---')
    for epoch in range(epochs):
        model.train()
        running_loss = 0.0
        for images, labels in train_loader:
            images, labels = images.to(device), labels.to(device)
            optimizer.zero_grad()
            outputs = model(images)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
            running_loss += loss.item()
        
        # Validation
        model.eval()
        correct, total = 0, 0
        with torch.no_grad():
            for images, labels in test_loader:
                images, labels = images.to(device), labels.to(device)
                outputs = model(images)
                _, preds = torch.max(outputs, 1)
                total += labels.size(0)
                correct += (preds == labels).sum().item()
        acc = 100 * correct / total
        print(f'Epoch {epoch+1}/{epochs} | Loss: {running_loss/len(train_loader):.4f} | Val Acc: {acc:.2f}%')
        
        if acc > best_acc:
            best_acc = acc
            torch.save(model.state_dict(), f'brain_tumor_{name}.pth')
    print(f'Finished {name}. Best Acc: {best_acc:.2f}%')
    return f'brain_tumor_{name}.pth'
#%% md 
## 4. Train The Heavy Models! 
#%% 
# Train ResNet18 (takes about 5-10 minutes on T4 GPU)
resnet_model = get_resnet()
train_model(resnet_model, 'resnet18', epochs=20)

# Train VGG16 (takes about 10-15 minutes on T4 GPU)
vgg_model = get_vgg()
train_model(vgg_model, 'vgg16', epochs=20)
#%% md 
## 5. Evaluation and Chart Generation 
#%% 
def evaluate(model_obj, path, name):
    if not os.path.exists(path):
        print(f'Skipping {name}, weights not found.')
        return None
    model_obj.load_state_dict(torch.load(path, map_location=device))
    model_obj = model_obj.to(device)
    model_obj.eval()
    
    all_labels = []
    all_preds = []
    with torch.no_grad():
        for images, labels in test_loader:
            outputs = model_obj(images.to(device))
            _, preds = torch.max(outputs, 1)
            all_labels.extend(labels.cpu().numpy())
            all_preds.extend(preds.cpu().numpy())
            
    acc = accuracy_score(all_labels, all_preds) * 100
    f1 = f1_score(all_labels, all_preds, average='weighted') * 100
    prec = precision_score(all_labels, all_preds, average='weighted') * 100
    rec = recall_score(all_labels, all_preds, average='weighted') * 100
    cm = confusion_matrix(all_labels, all_preds)
    
    # Plot Confusion Matrix
    plt.figure(figsize=(6,5))
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', xticklabels=class_names, yticklabels=class_names)
    plt.title(f'{name} Confusion Matrix')
    plt.savefig(f'{name}_cm.png')
    plt.close()
    
    return {'Accuracy': acc, 'F1': f1, 'Precision': prec, 'Recall': rec}

# PLEASE UPLOAD YOUR brain_tumor_cnn_v2.pth TO COLAB BEFORE RUNNING THIS CELL
metrics = {}
metrics['Custom_CNN'] = evaluate(BrainTumorCNN(), 'brain_tumor_cnn_v2.pth', 'Custom_CNN')
metrics['ResNet18'] = evaluate(get_resnet(), 'brain_tumor_resnet18.pth', 'ResNet18')
metrics['VGG16'] = evaluate(get_vgg(), 'brain_tumor_vgg16.pth', 'VGG16')

# Filter out None
metrics = {k: v for k, v in metrics.items() if v is not None}

# Save metrics to JSON
with open('model_metrics.json', 'w') as f:
    json.dump(metrics, f)
print('\nMetrics saved to model_metrics.json!')
print(json.dumps(metrics, indent=2))
