main_local.py 
""" 
KLERT MONITORING agent API - LOCAL DATASET VARIANT. 
 
Use this instead of MONITORING/api/main.py when the DICOM dataset already 
lives on the same machine as the API server. Clients send the relative 
folder path to a series instead of uploading a zip; the server reads it 
straight off disk. 
 
Same auth model as main.py: every endpoint needs an X-API-Key header, 
checked against the per-agent env vars in MONITORING/api/security.py. 
 
Setup: 
    1. Put the dataset somewhere on this machine, e.g. /data/Brain-Tumor-Progression 
    2. Add to .env:  DATASET_ROOT=/data/Brain-Tumor-Progression 
    3. Run:  uvicorn MONITORING.api.main_local:app --reload --port 8000 
"""

from typing import Optional

from fastapi import Depends, FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel

from MONITORING.api.security import require_agent_key

try:
    from MONITORING.api.config import resolve_dataset_path
except ImportError:
    from config import resolve_dataset_path

app = FastAPI(
    title="DeepCortex System Monitoring Agent API (local dataset)",
    version="1.0.0",
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


# --------------------------------------------------------------------------
# 1. Full pipeline (orchestrator)
# --------------------------------------------------------------------------

class PipelineRequest(BaseModel):
    patient_id: str
    baseline_mri_path: str      # e.g. "PGBM-019/10-02-1991-.../11.000000-T1post-35950"
    followup_mri_path: str
    baseline_mask_path: str
    followup_mask_path: str


@app.post("/agents/pipeline/run", dependencies=[Depends(require_agent_key("pipeline"))])
async def run_pipeline(body: PipelineRequest):
    try:
        from MONITORING.tools.dicom_loader import load_dicom_series
        from MONITORING.tools.mask_loader import load_mask_series
        from MONITORING.tools.registration import register_followup_to_baseline
        from MONITORING.tools.mask_registration import transform_mask_to_baseline
        from MONITORING.agents.orchestrator import app as pipeline_app

        b_mri = load_dicom_series(str(resolve_dataset_path(body.baseline_mri_path)))
        f_mri = load_dicom_series(str(resolve_dataset_path(body.followup_mri_path)))
        b_mask = load_mask_series(str(resolve_dataset_path(body.baseline_mask_path)))
        f_mask = load_mask_series(str(resolve_dataset_path(body.followup_mask_path)))

        # Perform registration to create registered_followup_mask
        try:
            _, reg_transform = register_followup_to_baseline(b_mri, f_mri)
            reg_f_mask = transform_mask_to_baseline(f_mask, b_mri, reg_transform)
        except Exception as reg_err:
            print(f"Registration warning (using raw followup_mask): {reg_err}")
            reg_f_mask = f_mask

        state = {
            "patient_id": body.patient_id,
            "baseline_mri": b_mri,
            "followup_mri": f_mri,
            "baseline_mask": b_mask,
            "followup_mask": f_mask,
            "registered_followup_mask": reg_f_mask,
        }
        result = pipeline_app.invoke(state)

        return {
            "monitoring_report": result.get("monitoring_report"),
            "baseline_visualization": result.get("baseline_visualization"),
            "followup_visualization": result.get("followup_visualization"),
            "difference_visualization": result.get("difference_visualization"),
            "status": result.get("status")
        }
    except Exception as exc:
        raise HTTPException(status_code=500, detail=f"Pipeline failed: {exc}")


# --------------------------------------------------------------------------
# 2. Mask analysis agent
# --------------------------------------------------------------------------

class MaskAnalysisRequest(BaseModel):
    patient_id: str
    baseline_mask_path: str
    followup_mask_path: str


@app.post("/agents/mask-analysis", dependencies=[Depends(require_agent_key("mask_analysis"))])
async def run_mask_analysis(body: MaskAnalysisRequest):
    from MONITORING.tools.mask_loader import load_mask_series
    from MONITORING.agents.mask_analysis_agent import mask_analysis_agent

    try:
        state = {
            "patient_id": body.patient_id,
            "baseline_mask": load_mask_series(str(resolve_dataset_path(body.baseline_mask_path))),
            "followup_mask": load_mask_series(str(resolve_dataset_path(body.followup_mask_path))),
        }
        return mask_analysis_agent(state)
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))


# --------------------------------------------------------------------------
# 3. Comparison agent
# --------------------------------------------------------------------------

class ComparisonRequest(BaseModel):
    patient_id: str
    baseline_mask_path: str
    followup_mask_path: str
    registered_followup_mask_path: Optional[str] = None


@app.post("/agents/comparison", dependencies=[Depends(require_agent_key("comparison"))])
async def run_comparison(body: ComparisonRequest):
    from MONITORING.tools.mask_loader import load_mask_series
    from MONITORING.agents.comparison_agent import comparison_agent

    try:
        f_mask = load_mask_series(str(resolve_dataset_path(body.followup_mask_path)))
        reg_path = body.registered_followup_mask_path
        reg_mask = load_mask_series(str(resolve_dataset_path(reg_path))) if reg_path else f_mask

        state = {
            "patient_id": body.patient_id,
            "baseline_mask": load_mask_series(str(resolve_dataset_path(body.baseline_mask_path))),
            "followup_mask": f_mask,
            "registered_followup_mask": reg_mask,
        }
        return comparison_agent(state)
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))
