def build_monitoring_graph():

    workflow = StateGraph(
        MonitoringState
    )

    # ----------------------------------------------
    # Add agents
    # ----------------------------------------------

    workflow.add_node(
        "mri_loader",
        mri_loader_agent
    )

    workflow.add_node(
        "mask_analysis",
        mask_analysis_agent
    )

    workflow.add_node(
        "comparison",
        comparison_node
    )

    workflow.add_node(
        "brain_region",
        brain_region_node
    )

    workflow.add_node(
        "visualization",
        visualization_agent
    )

    workflow.add_node(
        "report",
        report_agent
    )

    # ----------------------------------------------
    # Entry point
    # ----------------------------------------------

    workflow.set_entry_point(
        "mri_loader"
    )

    # ----------------------------------------------
    # Workflow
    # ----------------------------------------------

    workflow.add_edge(
        "mri_loader",
        "mask_analysis"
    )

    workflow.add_edge(
        "mask_analysis",
        "comparison"
    )

    workflow.add_edge(
        "comparison",
        "brain_region"
    )

    workflow.add_edge(
        "brain_region",
        "visualization"
    )

    workflow.add_edge(
        "visualization",
        "report"
    )

    workflow.add_edge(
        "report",
        END
    )

    # ----------------------------------------------
    # Compile
    # ----------------------------------------------

    return workflow.compile()

import os
import SimpleITK as sitk


def load_dicom_series(folder_path):
    """ 
    Load a DICOM series from a folder. 
 
    Parameters 
    ---------- 
    folder_path : str 
        Path to a folder containing DICOM slices. 
 
    Returns 
    ------- 
    image : SimpleITK.Image 
        Loaded 3D MRI volume. 
    """

    if not os.path.isdir(folder_path):
        raise FileNotFoundError(
            f"DICOM folder does not exist: {folder_path}"
        )

    reader = sitk.ImageSeriesReader()

    series_ids = reader.GetGDCMSeriesIDs(folder_path)

    if not series_ids:
        raise ValueError(
            f"No DICOM series found in: {folder_path}"
        )

    # Use the first series found.
    series_id = series_ids[0]

    dicom_files = reader.GetGDCMSeriesFileNames(
        folder_path,
        series_id
    )

    if not dicom_files:
        raise ValueError(
            f"No DICOM files found in: {folder_path}"
        )

    reader.SetFileNames(dicom_files)

    image = reader.Execute()

    return image

import numpy as np
import SimpleITK as sitk


def validate_mask(mask, name):
    """ 
    Validate a tumor mask. 
    """

    array = sitk.GetArrayFromImage(mask)

    unique_values = np.unique(array)

    tumor_voxels = np.count_nonzero(array)

    is_empty = tumor_voxels == 0

    return {
        "name": name,
        "size": mask.GetSize(),
        "spacing": mask.GetSpacing(),
        "origin": mask.GetOrigin(),
        "direction": mask.GetDirection(),
        "unique_values": unique_values.tolist(),
        "tumor_voxels": int(tumor_voxels),
        "is_empty": is_empty,
        "valid": not is_empty
    }


def mask_analysis_agent(state):

    print("Mask Analysis Agent started")

    baseline_mask = state["baseline_mask"]
    followup_mask = state["followup_mask"]

    baseline_info = validate_mask(
        baseline_mask,
        "Baseline Mask"
    )

    followup_info = validate_mask(
        followup_mask,
        "Follow-up Mask"
    )

    print("\nBaseline Mask")
    print("Size:", baseline_info["size"])
    print("Spacing:", baseline_info["spacing"])
    print("Tumor voxels:", baseline_info["tumor_voxels"])
    print("Unique values:", baseline_info["unique_values"])

    print("\nFollow-up Mask")
    print("Size:", followup_info["size"])
    print("Spacing:", followup_info["spacing"])
    print("Tumor voxels:", followup_info["tumor_voxels"])
    print("Unique values:", followup_info["unique_values"])

    return {
        "baseline_mask_info": baseline_info,
        "followup_mask_info": followup_info,
        "status": "Mask analysis completed"
    }

import SimpleITK as sitk


def register_followup_to_baseline(
    baseline_mri,
    followup_mri
):
    """ 
    Rigidly register follow-up MRI to baseline MRI. 
 
    Baseline MRI is the fixed/reference image. 
    Follow-up MRI is the moving image. 
    """

    fixed = sitk.Cast(
        baseline_mri,
        sitk.sitkFloat32
    )

    moving = sitk.Cast(
        followup_mri,
        sitk.sitkFloat32
    )

    # Initial transformation based on image geometry
    initial_transform = sitk.CenteredTransformInitializer(
        fixed,
        moving,
        sitk.Euler3DTransform(),
        sitk.CenteredTransformInitializerFilter.GEOMETRY
    )

    registration = sitk.ImageRegistrationMethod()

    # Similarity metric
    registration.SetMetricAsMattesMutualInformation(
        numberOfHistogramBins=50
    )

    # Sampling improves speed
    registration.SetMetricSamplingStrategy(
        registration.RANDOM
    )

    registration.SetMetricSamplingPercentage(
        0.10
    )

    # Interpolator for MRI intensity images
    registration.SetInterpolator(
        sitk.sitkLinear
    )

    # Optimizer
    registration.SetOptimizerAsGradientDescent(
        learningRate=1.0,
        numberOfIterations=100,
        convergenceMinimumValue=1e-6,
        convergenceWindowSize=10
    )

    registration.SetOptimizerScalesFromPhysicalShift()

    registration.SetInitialTransform(
        initial_transform,
        inPlace=False
    )

    final_transform = registration.Execute(
        fixed,
        moving
    )

    # Resample follow-up MRI into baseline space
    registered_followup = sitk.Resample(
        moving,
        fixed,
        final_transform,
        sitk.sitkLinear,
        0.0,
        moving.GetPixelID()
    )

    return registered_followup, final_transform

import numpy as np
import SimpleITK as sitk


def calculate_new_regions(
    baseline_mask,
    registered_followup_mask
):
    """ 
    Identify tumor regions that appear in the 
    follow-up scan but were not present at baseline. 
    """

    baseline = sitk.GetArrayFromImage(
        baseline_mask
    ).astype(bool)

    followup = sitk.GetArrayFromImage(
        registered_followup_mask
    ).astype(bool)

    # New tumor = follow-up tumor AND NOT baseline tumor
    new_region = followup & (~baseline)

    new_region_image = sitk.GetImageFromArray(
        new_region.astype(np.uint8)
    )

    # Keep baseline spatial information
    new_region_image.CopyInformation(
        baseline_mask
    )

    new_voxels = int(np.count_nonzero(new_region))

    spacing = baseline_mask.GetSpacing()

    voxel_volume_mm3 = (
        spacing[0]
        * spacing[1]
        * spacing[2]
    )

    new_volume_mm3 = (
        new_voxels * voxel_volume_mm3
    )

    new_volume_cm3 = (
        new_volume_mm3 / 1000.0
    )

    # Connected components
    connected = sitk.ConnectedComponent(
        new_region_image
    )

    statistics = sitk.LabelShapeStatisticsImageFilter()
    statistics.Execute(connected)

    number_of_regions = statistics.GetNumberOfLabels()

    return {
        "new_region_mask": new_region_image,
        "new_voxels": new_voxels,
        "new_volume_mm3": float(new_volume_mm3),
        "new_volume_cm3": float(new_volume_cm3),
        "number_of_new_regions": int(number_of_regions)
    }

import SimpleITK as sitk
import numpy as np


def calculate_tumor_centroid(mask):

    array = sitk.GetArrayFromImage(mask)

    # Find tumor voxels
    coordinates = np.argwhere(array > 0)

    if coordinates.size == 0:
        return None

    # SimpleITK array coordinates are:
    # [z, y, x]
    centroid_zyx = coordinates.mean(axis=0)

    z = float(centroid_zyx[0])
    y = float(centroid_zyx[1])
    x = float(centroid_zyx[2])

    # Convert index coordinates to physical coordinates
    centroid_physical = mask.TransformContinuousIndexToPhysicalPoint(
        (x, y, z)
    )

    return centroid_physical


def calculate_centroid_shift(
    baseline_mask,
    registered_followup_mask
):

    baseline_centroid = calculate_tumor_centroid(
        baseline_mask
    )

    followup_centroid = calculate_tumor_centroid(
        registered_followup_mask
    )

    if baseline_centroid is None:
        raise ValueError(
            "Baseline mask contains no tumor."
        )

    if followup_centroid is None:
        raise ValueError(
            "Follow-up mask contains no tumor."
        )

    shift_vector = np.array(
        followup_centroid
    ) - np.array(
        baseline_centroid
    )

    shift_distance = float(
        np.linalg.norm(shift_vector)
    )

    return {
        "baseline_centroid": baseline_centroid,
        "followup_centroid": followup_centroid,
        "shift_vector_mm": tuple(
            shift_vector.tolist()
        ),
        "shift_distance_mm": shift_distance
    }

import numpy as np
import SimpleITK as sitk
from skimage.measure import marching_cubes, mesh_surface_area


def calculate_surface_area(mask):
    """ 
    Calculate 3D tumor surface area. 
 
    Returns: 
        surface_area_mm2 
        surface_area_cm2 
    """

    # Convert SimpleITK image to NumPy
    array = sitk.GetArrayFromImage(mask)

    # Binary tumor mask
    binary = (array > 0).astype(np.uint8)

    # Check for empty mask
    if np.count_nonzero(binary) == 0:
        return {
            "surface_area_mm2": 0.0,
            "surface_area_cm2": 0.0
        }

    # SimpleITK -> NumPy changes axis order:
    # NumPy = Z, Y, X
    #
    # Therefore spacing must also be:
    # Z, Y, X
    spacing = mask.GetSpacing()

    voxel_spacing = (
        spacing[2],
        spacing[1],
        spacing[0]
    )

    # Extract 3D surface
    vertices, faces, normals, values = marching_cubes(
        binary,
        level=0.5,
        spacing=voxel_spacing
    )

    # Calculate surface area
    surface_area_mm2 = mesh_surface_area(
        vertices,
        faces
    )

    surface_area_cm2 = (
        surface_area_mm2 / 100.0
    )

    return {
        "surface_area_mm2": float(surface_area_mm2),
        "surface_area_cm2": float(surface_area_cm2)
    }

import math


def calculate_sphericity(volume_mm3, surface_area_mm2):
    """ 
    Calculate 3D tumor sphericity. 
 
    Sphericity: 
        1.0 = perfect sphere 
        lower values = increasingly irregular shape 
    """

    if volume_mm3 <= 0 or surface_area_mm2 <= 0:
        return None

    sphericity = (
        (math.pi ** (1.0 / 3.0))
        * ((6.0 * volume_mm3) ** (2.0 / 3.0))
        / surface_area_mm2
    )

    return float(sphericity)


def classify_shape_irregularity(sphericity):

    if sphericity is None:
        return "undefined"

    if sphericity >= 0.90:
        return "low irregularity"

    elif sphericity >= 0.75:
        return "moderate irregularity"

    else:
        return "high irregularity"


def calculate_shape_metrics(
    volume_mm3,
    surface_area_mm2
):

    sphericity = calculate_sphericity(
        volume_mm3,
        surface_area_mm2
    )

    classification = classify_shape_irregularity(
        sphericity
    )

    return {
        "sphericity": sphericity,
        "irregularity": classification
    }

def calculate_boundary_change(
    baseline_surface_mm2,
    followup_surface_mm2
):

    # Absolute change in tumor boundary
    boundary_change_mm2 = (
        followup_surface_mm2 -
        baseline_surface_mm2
    )

    # Percentage change
    if baseline_surface_mm2 == 0:

        boundary_change_percentage = 0.0

    else:

        boundary_change_percentage = (
            boundary_change_mm2 /
            baseline_surface_mm2
        ) * 100.0

    return {

        "boundary_change_mm2":
            float(boundary_change_mm2),

        "boundary_change_percentage":
            float(boundary_change_percentage)
    }

import numpy as np

def calculate_confidence_score(
    baseline_mri,
    followup_mri,
    baseline_mask,
    followup_mask,
    registered_followup_mask
):

    checks = {}

    # ==================================================
    # 1. BASELINE MRI / MASK DIMENSIONS
    # ==================================================

    baseline_dimension_check = (
        baseline_mri.GetSize()
        ==
        baseline_mask.GetSize()
    )

    checks["baseline_dimensions"] = (
        baseline_dimension_check
    )

    # ==================================================
    # 2. FOLLOW-UP MRI / MASK DIMENSIONS
    # ==================================================

    followup_dimension_check = (
        followup_mri.GetSize()
        ==
        followup_mask.GetSize()
    )

    checks["followup_dimensions"] = (
        followup_dimension_check
    )

    # ==================================================
    # 3. BASELINE SPACING
    # ==================================================

    baseline_spacing_check = np.allclose(
        baseline_mri.GetSpacing(),
        baseline_mask.GetSpacing(),
        atol=0.01
    )

    checks["baseline_spacing"] = (
        baseline_spacing_check
    )

    # ==================================================
    # 4. FOLLOW-UP SPACING
    # ==================================================

    followup_spacing_check = np.allclose(
        followup_mri.GetSpacing(),
        followup_mask.GetSpacing(),
        atol=0.01
    )

    checks["followup_spacing"] = (
        followup_spacing_check
    )

    # ==================================================
    # 5. BASELINE MASK VALIDITY
    # ==================================================

    baseline_array = sitk.GetArrayFromImage(
        baseline_mask
    )

    baseline_unique = np.unique(
        baseline_array
    )

    baseline_mask_binary = np.all(
        np.isin(
            baseline_unique,
            [0, 1]
        )
    )

    checks["baseline_mask_binary"] = (
        baseline_mask_binary
    )

    # ==================================================
    # 6. FOLLOW-UP MASK VALIDITY
    # ==================================================

    followup_array = sitk.GetArrayFromImage(
        followup_mask
    )

    followup_unique = np.unique(
        followup_array
    )

    followup_mask_binary = np.all(
        np.isin(
            followup_unique,
            [0, 1]
        )
    )

    checks["followup_mask_binary"] = (
        followup_mask_binary
    )

    # ==================================================
    # 7. TUMOR PRESENT
    # ==================================================

    baseline_tumor_present = (
        np.count_nonzero(
            baseline_array
        ) > 0
    )

    followup_tumor_present = (
        np.count_nonzero(
            followup_array
        ) > 0
    )

    checks["baseline_tumor_present"] = (
        baseline_tumor_present
    )

    checks["followup_tumor_present"] = (
        followup_tumor_present
    )

    # ==================================================
    # 8. REGISTERED MASK VALIDITY
    # ==================================================

    registered_array = sitk.GetArrayFromImage(
        registered_followup_mask
    )

    registered_unique = np.unique(
        registered_array
    )

    registered_mask_binary = np.all(
        np.isin(
            registered_unique,
            [0, 1]
        )
    )

    checks["registered_mask_binary"] = (
        registered_mask_binary
    )

    # ==================================================
    # 9. REGISTERED MASK SIZE
    # ==================================================

    registration_dimension_check = (
        baseline_mask.GetSize()
        ==
        registered_followup_mask.GetSize()
    )

    checks["registration_dimensions"] = (
        registration_dimension_check
    )

    # ==================================================
    # SCORE
    # ==================================================

    passed_checks = sum(
        checks.values()
    )

    total_checks = len(checks)

    confidence_score = (
        passed_checks /
        total_checks
    ) * 100.0

    # ==================================================
    # CONFIDENCE LEVEL
    # ==================================================

    if confidence_score >= 90:

        confidence_level = "High"

    elif confidence_score >= 70:

        confidence_level = "Moderate"

    else:

        confidence_level = "Low"

    return {

        "confidence_score":
            float(confidence_score),

        "confidence_level":
            confidence_level,

        "passed_checks":
            passed_checks,

        "total_checks":
            total_checks,

        "checks":
            checks
    }
